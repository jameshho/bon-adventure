extends Control
# Cached 2D projection of the actual town meshes. No second 3D render pass.
var full_map := false
var MAP_SIZE := Vector2(208,208)
var ZOOM := 2.4
var world: Node3D
var player: Node3D
var surfaces: Array[Dictionary] = []
var landmarks: Array[Dictionary] = []
var map_position := Vector2.ZERO
var map_yaw := 0.0
var elapsed := 0.0

func setup(town: Node3D, explorer: Node3D) -> void:
    world = town
    player = explorer
    name = "TownMinimap"
    position = Vector2(16,16)
    size = MAP_SIZE
    clip_contents = true
    mouse_filter = Control.MOUSE_FILTER_IGNORE
    rebuild()
    update_pose()

func point2(p: Vector3) -> Vector2:
    return Vector2(p.x,p.z)

func update_pose() -> void:
    if full_map:
        position=Vector2(30,60)
        size=get_viewport_rect().size-Vector2(60,120)
        MAP_SIZE=size
        ZOOM=minf(size.x/170.0,size.y/190.0)
        map_position=Vector2(0,75)
        map_yaw=0
        return
    map_position = point2(world.to_local(player.global_position))
    var forward: Vector3 = world.global_basis.inverse() * -player.global_basis.z
    map_yaw = atan2(-forward.x,-forward.z)

func project(point: Vector2) -> Vector2:
    return MAP_SIZE * .5 + (point-map_position).rotated(map_yaw)*ZOOM

func footprint(node: Node3D) -> PackedVector2Array:
    var low := Vector2(INF,INF)
    var high := Vector2(-INF,-INF)
    for mesh in node.find_children("*","MeshInstance3D",true,false):
        if not mesh.mesh: continue
        var bounds: AABB = mesh.get_aabb()
        for i in range(8):
            var point := point2(node.to_local(mesh.to_global(bounds.get_endpoint(i))))
            low = low.min(point)
            high = high.max(point)
    if not low.is_finite(): return PackedVector2Array()
    return PackedVector2Array([low,Vector2(high.x,low.y),high,Vector2(low.x,high.y)])

# Call after structural mesh edits. Building transforms are read live, so relocating
# an existing building does not require a separate map-coordinate edit.
func rebuild() -> void:
    surfaces.clear()
    landmarks.clear()
    for node in world.get_children():
        if not node is MeshInstance3D or not node.mesh: continue
        var label := str(node.name)
        var filled: bool = label in ["Coastal shallows","Island beaches","Natural town terrain","Park pond"]
        var ribbon: bool = label.begins_with("Winding park path") or label in ["Curving neighborhood road","Loop road sidewalks","Shop promenade","Civic promenade","Bakery approach"]
        if not filled and not ribbon: continue
        var arrays: Array = node.mesh.surface_get_arrays(0)
        var vertices: PackedVector3Array = arrays[Mesh.ARRAY_VERTEX]
        var points := PackedVector2Array()
        var width := 1.0
        if ribbon:
            width = vertices[0].distance_to(vertices[1])
            for i in range(0,vertices.size()-1,6):
                points.append(point2(world.to_local(node.to_global((vertices[i]+vertices[i+1])*.5))))
            var last := vertices.size()-2
            points.append(point2(world.to_local(node.to_global((vertices[last]+vertices[last+1])*.5))))
        else:
            for i in range(0,vertices.size(),3): points.append(point2(world.to_local(node.to_global(vertices[i]))))
        var color: Color = node.material_override.albedo_color
        surfaces.append({"points":points,"fill":filled,"width":width,"color":color})
    for node in world.get_node("Buildings").get_children():
        if node is Node3D:
            landmarks.append({"node":node,"points":footprint(node),"color":Color("f2dbc0")})
    for title in ["Supermarket", "Library", "Bank", "Police Station", "Chapel"]:
        var roof := world.get_node_or_null(title + " roof")
        if roof: landmarks.append({"node":roof,"points":footprint(roof),"color":Color("d5dfba")})
    var home := world.get_node_or_null("HouseStructure")
    if home: landmarks.append({"node":home,"points":PackedVector2Array([Vector2(-11,-9),Vector2(12,-9),Vector2(12,12),Vector2(-11,12)]),"color":Color("dca38d")})
    var zoo := world.get_node_or_null("ParkZoo")
    if zoo:
        for data in [[76.0,7.0],[59.0,5.0]]:
            landmarks.append({"node":zoo,"points":PackedVector2Array([Vector2(-40,data[0]-data[1]),Vector2(-28,data[0]-data[1]),Vector2(-28,data[0]+data[1]),Vector2(-40,data[0]+data[1])]),"color":Color("cab880")})
    queue_redraw()

func _process(delta: float) -> void:
    if not is_instance_valid(player) or not is_instance_valid(world): return
    elapsed += delta
    if elapsed < 1.0/30.0: return
    elapsed = 0
    update_pose()
    queue_redraw()

func _draw() -> void:
    draw_rect(Rect2(Vector2.ZERO,MAP_SIZE),Color("4f9eae"))
    for surface in surfaces:
        var polygon := PackedVector2Array()
        for p in surface.points: polygon.append(project(p))
        if polygon.size() < 3: continue
        if surface.fill: draw_colored_polygon(polygon,surface.color)
        else: draw_polyline(polygon,surface.color,surface.width*ZOOM,true)
    for landmark in landmarks:
        if not is_instance_valid(landmark.node): continue
        var polygon := PackedVector2Array()
        for p in landmark.points:
            polygon.append(project(point2(world.to_local(landmark.node.to_global(Vector3(p.x,0,p.y))))))
        if polygon.size()>2:
            draw_colored_polygon(polygon,landmark.color)
            polygon.append(polygon[0])
            draw_polyline(polygon,Color("566d69"),1.3,true)
    if full_map:
        var font:=ThemeDB.fallback_font
        for data in [["SUPERMARKET",Vector2(-33,43)],["BANK",Vector2(25,45)],["POLICE",Vector2(43,45)],["HOME",Vector2(0,9)],["LOW CLOUD",Vector2(18,22)],["HOSPITAL / HELIPAD",Vector2(-32,116)],["SCHOOL",Vector2(-4,116)],["CHAPEL",Vector2(46.5,75)]]:
            var location:Vector2=project(data[1])
            draw_circle(location,3,Color("425c66"))
            draw_string(font,location+Vector2(-20,-7),data[0],HORIZONTAL_ALIGNMENT_LEFT,-1,12,Color("253f4c"))
    var center := project(point2(world.to_local(player.global_position))) if full_map else MAP_SIZE*.5
    draw_circle(center,10,Color("203a48"))
    draw_colored_polygon(PackedVector2Array([center+Vector2(0,-9),center+Vector2(6,7),center+Vector2(0,4),center+Vector2(-6,7)]),Color("fff8de"))
    draw_rect(Rect2(Vector2.ONE,MAP_SIZE-Vector2.ONE*2),Color("fff1d6"),false,2)
